-- 1. FIRST: Define helper functions that everything else depends on
local function get_ui_parent()
    local success, parent = pcall(function() return gethui and gethui() end)
    if success and parent then return parent end
    success, parent = pcall(function() return game:GetService("CoreGui") end)
    if success and parent then return parent end
    return game:GetService("Players").LocalPlayer:WaitForChild("PlayerGui")
end

}
library.__index = library

local connections = {}
local function track_connection(conn)
    if conn then table.insert(connections, conn) end
    return conn
end

-- 3. THIRD: Define the Unload/Cleanup logic
function library:Unload()
    for _, conn in ipairs(connections) do
        if conn then conn:Disconnect() end
    end
    table.clear(connections)
    table.clear(theme_bindings)
    self._onThemeChanged = nil

    local parent = get_ui_parent()
    if parent:FindFirstChild("MonolithUI") then parent.MonolithUI:Destroy() end
    if parent:FindFirstChild("MonolithNotifs") then parent.MonolithNotifs:Destroy() end
    for _, v in ipairs(parent:GetChildren()) do
        if v.Name:match("^MobileBtn_") then v:Destroy() end
    end

    print("Unloaded Successfully")
end

local function global_cleanup()
    local parent = get_ui_parent()
    if parent:FindFirstChild("MonolithUI") then parent.MonolithUI:Destroy() end
    if parent:FindFirstChild("MonolithNotifs") then parent.MonolithNotifs:Destroy() end
end

local theme_bindings = {}

local function push_binding(inst, apply)
    local b = { inst = inst, apply = apply, alive = true }
    table.insert(theme_bindings, b)
    if inst and inst.Destroying then
        inst.Destroying:Connect(function() b.alive = false end)
    end
    apply(true)
    return b
end

local TokenTag = {}
local function T(key, mutator)
    return setmetatable({ key = key, mutator = mutator }, TokenTag)
end
local function is_token(v) return type(v) == "table" and getmetatable(v) == TokenTag end
library.T = T

function library.lighten(c, amt)
    amt = amt or 0.15
    return Color3.new(c.R + (1 - c.R) * amt, c.G + (1 - c.G) * amt, c.B + (1 - c.B) * amt)
end
function library.darken(c, amt)
    amt = amt or 0.15
    return Color3.new(c.R * (1 - amt), c.G * (1 - amt), c.B * (1 - amt))
end
function library.resolve(v)
    if is_token(v) then
        local c = Theme[v.key] or rgb(255, 0, 255)
        if v.mutator then c = v.mutator(c) end
        return c
    end
    return v
end

function library:bind(inst, prop, resolver, animate)
    local b
    b = push_binding(inst, function(instant)
        local ok, val = pcall(resolver)
        if not ok or val == nil then return end
        if instant or animate == false then
            pcall(function() inst[prop] = val end)
        else
            tween_service:Create(inst, TweenInfo.new(0.25, Enum.EasingStyle.Quint), {[prop] = val}):Play()
        end
    end)
    b.update = function(_, instant) if b.alive then b.apply(instant) end end
    return b
end

function library:bindFn(inst, fn)
    local b = push_binding(inst, function(instant) pcall(fn, instant) end)
    b.update = function(_, instant) if b.alive then b.apply(instant) end end
    return b
end

function library:RefreshTheme(instant)
    for i = #theme_bindings, 1, -1 do
        local b = theme_bindings[i]
        if not b.alive then table.remove(theme_bindings, i) else b.apply(instant) end
    end
    if self._onThemeChanged then
        for _, fn in ipairs(self._onThemeChanged) do pcall(fn, Theme) end
    end
end

function library:OnThemeChanged(fn)
    self._onThemeChanged = self._onThemeChanged or {}
    table.insert(self._onThemeChanged, fn)
end

function library:SetThemeColor(key, color, instant)
    if typeof(color) ~= "Color3" then return end
    Theme[key] = color
    self.CurrentPreset = "Custom"
    self:RefreshTheme(instant)
end

function library:SetTheme(tbl, instant)
    for _, k in ipairs(THEME_KEYS) do
        local v = tbl[k]
        if typeof(v) == "Color3" then Theme[k] = v end
    end
    self:RefreshTheme(instant)
end

function library:ApplyPreset(name, instant)
    local p = Presets[name]
    if not p then return false end
    self:SetTheme(p, instant)
    self.CurrentPreset = name
    return true
end

function library:GetTheme()
    local out = {}
    for _, k in ipairs(THEME_KEYS) do out[k] = Theme[k] end
    return out
end

function library:ExportTheme()
    local out = { _preset = self.CurrentPreset }
    for _, k in ipairs(THEME_KEYS) do
        local c = Theme[k]
        out[k] = { math.floor(c.R * 255 + 0.5), math.floor(c.G * 255 + 0.5), math.floor(c.B * 255 + 0.5) }
    end
    return http_service:JSONEncode(out)
end

function library:ImportTheme(json)
    local ok, data = pcall(function() return http_service:JSONDecode(json) end)
    if not ok or type(data) ~= "table" then return false end
    local tbl = {}
    for _, k in ipairs(THEME_KEYS) do
        local v = data[k]
        if type(v) == "table" and #v >= 3 then tbl[k] = rgb(v[1], v[2], v[3]) end
    end
    self:SetTheme(tbl)
    self.CurrentPreset = data._preset or "Custom"
    return true
end

function library:tween(obj, props, time)
    local resolved = {}
    for k, v in pairs(props) do resolved[k] = library.resolve(v) end
    local t = tween_service:Create(obj, TweenInfo.new(time or 0.2, Enum.EasingStyle.Quint, Enum.EasingDirection.Out), resolved)
    t:Play()
    return t
end

function library:create(class, props)
    local ins = Instance.new(class)
    local tokens = {}
    for k, v in pairs(props) do
        if is_token(v) then
            tokens[k] = v
        else
            ins[k] = v
        end
    end
    for prop, tok in pairs(tokens) do
        library:bind(ins, prop, function() return library.resolve(tok) end)
    end
    return ins
end

function library:draggify(frame, drag_area)
    local dragging, startPos, startInput; -- Syntax error fixed
    (drag_area or frame).InputBegan:Connect(function(input)
        if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then
            if frame:GetAttribute("PositionLocked") then return end
            dragging = true; startInput = input.Position; startPos = frame.Position
        end
    end)
    -- TRACKED CONNECTION
    track_connection(uis.InputChanged:Connect(function(input)
        if dragging and (input.UserInputType == Enum.UserInputType.MouseMovement or input.UserInputType == Enum.UserInputType.Touch) then
            if frame:GetAttribute("PositionLocked") then dragging = false return end
            local delta = input.Position - startInput
            frame.Position = dim2(startPos.X.Scale, startPos.X.Offset + delta.X, startPos.Y.Scale, startPos.Y.Offset + delta.Y)
        end
    end))
    -- TRACKED CONNECTION
    track_connection(uis.InputEnded:Connect(function(input)
        if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then dragging = false end
    end))
end

local function color_icon(iconInstance, color, instant)
    if not iconInstance then return end
    color = library.resolve(color)
    if typeof(color) ~= "Color3" then return end
    if iconInstance:IsA("ImageLabel") or iconInstance:IsA("ImageButton") then
        if instant then iconInstance.ImageColor3 = color
        else tween_service:Create(iconInstance, TweenInfo.new(0.15), {ImageColor3 = color}):Play() end
    end
    for _, v in pairs(iconInstance:GetDescendants()) do
        if v:IsA("ImageLabel") or v:IsA("ImageButton") then
            if instant then v.ImageColor3 = color
            else tween_service:Create(v, TweenInfo.new(0.15), {ImageColor3 = color}):Play() end
        end
    end
end

local function get_icon(iconName, color)
    if not iconName then return nil end

    local function resolve_color()
        if type(color) == "function" then return color() end
        if type(color) == "string" then return Theme[color] or rgb(255,255,255) end
        return color or rgb(255, 255, 255)
    end

    local isAsset = iconName:match("^rbxassetid://") or iconName:match("^%d+$")
    if isAsset then
        local img = Instance.new("ImageLabel")
        img.BackgroundTransparency = 1
        img.Image = iconName:match("^%d+$") and "rbxassetid://" .. iconName or iconName
        library:bindFn(img, function(instant) color_icon(img, resolve_color(), instant) end)
        return img
    end

    local holder = Instance.new("Frame")
    holder.BackgroundTransparency = 1
    holder:SetAttribute("IsIconHolder", true)
    local binding = library:bindFn(holder, function(instant) color_icon(holder, resolve_color(), instant) end)
    
    -- Removed the invalid holder.__setTint line that caused the crash!

    task.spawn(function()
        local waited = 0
        while not Icons and waited < 5 do task.wait(0.1); waited = waited + 0.1 end
        if Icons then
            local iconStr = iconName:gsub("^lucide:", "")
            local ok, iconObj = pcall(function()
                return Icons.Image({ Icon = iconStr, Colors = { resolve_color() } })
            end)
            if ok and iconObj and iconObj.IconFrame then
                iconObj.IconFrame.BackgroundTransparency = 1
                iconObj.IconFrame.Size = UDim2.new(1, 0, 1, 0)
                iconObj.IconFrame.Parent = holder
                binding:update(true) 
            end
        end
    end)
    return holder
end

local function bind_icon(iconHolder, resolver)
    if not iconHolder then return end
    return library:bindFn(iconHolder, function(instant)
        color_icon(iconHolder, resolver(), instant)
    end)
end

local function BootSequence(windowFrame, windowName)
    local main = library:create("Frame", {
        Parent = windowFrame, Size = dim2(1, 0, 1, 0), BackgroundColor3 = rgb(0, 0, 0), 
        ZIndex = 1000, BorderSizePixel = 0
    })
    local logo = library:create("TextLabel", {
        Parent = main, Text = windowName, Position = dim2(0.5, 0, 0.4, 0), AnchorPoint = Vector2.new(0.5, 0.5),
        Size = dim2(0, 200, 0, 50), BackgroundTransparency = 1, TextColor3 = Theme.Text,
        FontFace = library.font, TextSize = 32, TextTransparency = 1, ZIndex = 1001
    })
    local barBg = library:create("Frame", {
        Parent = main, Size = dim2(0, 200, 0, 4), Position = dim2(0.5, 0, 0.6, 0), AnchorPoint = Vector2.new(0.5, 0.5),
        BackgroundColor3 = Theme.Outline, BorderSizePixel = 0, ZIndex = 1001
    })
    library:create("UICorner", {Parent = barBg})
    local barFill = library:create("Frame", {
        Parent = barBg, Size = dim2(0, 0, 1, 0), BackgroundColor3 = Theme.Accent, BorderSizePixel = 0, ZIndex = 1002
    })
    library:create("UICorner", {Parent = barFill})
    local status = library:create("TextLabel", {
        Parent = main, Text = "Initializing...", Position = dim2(0.5, 0, 0.65, 0), AnchorPoint = Vector2.new(0.5, 0.5),
        Size = dim2(0, 200, 0, 20), BackgroundTransparency = 1, TextColor3 = Theme.MutedText,
        FontFace = library.font, TextSize = 14, TextTransparency = 1, ZIndex = 1001
    })

    return coroutine.wrap(function()
        library:tween(logo, {TextTransparency = 0}, 1)
        task.wait(0.5)
        library:tween(status, {TextTransparency = 0}, 0.5)
        local steps = {"Loading Core...", "Fetching Assets...", "Applying Theme...", "Finalizing..."}
        for i, step in ipairs(steps) do
            status.Text = step
            library:tween(barFill, {Size = dim2(i/ #steps, 0, 1, 0)}, 0.5)
            task.wait(0.6)
        end
        library:tween(logo, {TextTransparency = 1}, 0.5)
        library:tween(status, {TextTransparency = 1}, 0.5)
        library:tween(barFill, {BackgroundTransparency = 1}, 0.3)
        library:tween(barBg, {BackgroundTransparency = 1}, 0.3)
        task.wait(0.5)
        local fadeOut = library:tween(main, {BackgroundTransparency = 1}, 1)
        fadeOut.Completed:Connect(function() main:Destroy() end)
        task.wait(1)
    end)
end

    function win:GetConfigData()
        local data = {}
        for key, el in pairs(self._elementRegistry) do
            if not el.NoSave and type(el.get_value) == "function" then
                local ok, v = pcall(function() return el:get_value() end)
                if ok and v ~= nil then data[key] = v end
            end
        end
        return data
    end

    function win:LoadConfigData(data)
        if type(data) ~= "table" then return end
        _G.IsLoadingConfig = true

        -- PASS 1 & 2: Colors first, then everything else
        local deferred = {}
        for name, value in pairs(data) do
            local el = self._elementRegistry[name] or self._legacyRegistry[name]
            if el and type(el.set) == "function" then
                if el._isColor then
                    pcall(function() el:set(value) end)
                else
                    table.insert(deferred, {el = el, value = value, name = name})
                end
            end
        end

        for _, item in ipairs(deferred) do
            local success, err = pcall(function() item.el:set(item.value) end)
            if not success then
                warn("[Config Error] '" .. tostring(item.name) .. "': " .. tostring(err))
            end
            if item.el.refresh then pcall(function() item.el:refresh() end) end
        end

        if self.RefreshMobileButtonStates then self:RefreshMobileButtonStates() end

        task.spawn(function()
            task.wait(0.2)
            _G.IsLoadingConfig = false 
        end)
    end

        bootCoroutine()
    end

    end

            function tog:set(val)
                local finalState
                local t = type(val)

                if t == "boolean" then finalState = val
                elseif t == "number" then finalState = val ~= 0
                elseif t == "string" then
                    local l = val:lower(); finalState = (l == "true" or l == "1" or l == "on" or l == "yes")
                elseif t == "table" then
                    if val.State ~= nil then finalState = val.State == true
                    elseif val.enabled ~= nil then finalState = val.enabled == true
                    elseif val.Value ~= nil then finalState = val.Value == true
                    else finalState = self.enabled end
                    
                    if val.Key then
                        local parsed = nil
                        pcall(function() parsed = Enum.KeyCode[val.Key] end)
                        if not parsed then pcall(function() parsed = Enum.UserInputType[val.Key] end) end
                        if val.Key == "ScrollUp" or val.Key == "ScrollDown" then parsed = val.Key end

                            boundKey = parsed; bindBtn.Text = typeof(parsed) == "EnumItem" and parsed.Name or parsed; bindState = "bound"

                            boundKey = nil; bindBtn.Text = "—"; bindState = "idle"
                        end
                    else
                        boundKey = nil; bindBtn.Text = "—"; bindState = "idle"
                    end
                elseif val == nil then finalState = self.enabled
                else finalState = val and true or false end

                self.enabled = finalState
                container.Visible = finalState
                if self._indBind then self._indBind:update() end
                if bindState then library:RefreshTheme() end
                
                if p.Callback then task.spawn(function() p.Callback(finalState) end) end
            end

                        else
                            key = input.KeyCode; btn.Text = "  " .. (p.Name or p.name or "Keybind") .. " : [" .. key.Name .. "]"
                            if p.Callback then p.Callback(key) end
                        end

            end))

        BackgroundTransparency = 1
    })
    library:create("UICorner", {Parent = notif, CornerRadius = dim(0, 6)})
    local stroke = library:create("UIStroke", {Parent = notif, Color = Theme.Outline, Thickness = 1, Transparency = 1})
    
    local title = library:create("TextLabel", {
        Parent = notif,
        Size = dim2(1, -20, 1, 0),
        Position = dim2(0, 10, 0, 0),
        BackgroundTransparency = 1,
        Text = name,
        TextColor3 = Theme.Text,
        FontFace = library.font,
        TextSize = 13,
        TextXAlignment = Enum.TextXAlignment.Left,
        TextTransparency = 1
    })
    
    library:tween(notif, {BackgroundTransparency = 0}, 0.3)
    library:tween(stroke, {Transparency = 0}, 0.3)
    library:tween(title, {TextTransparency = 0}, 0.3)
    
    task.delay(duration, function()
        local fade = library:tween(notif, {BackgroundTransparency = 1}, 0.5)
        library:tween(stroke, {Transparency = 1}, 0.5)
        library:tween(title, {TextTransparency = 1}, 0.5)
        fade.Completed:Connect(function()
            notif:Destroy()
        end)
    end)
end

return library, library
